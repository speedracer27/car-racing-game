<script>
let x=120, y=-50, cy=-150;
let score=0, playing=false, timer;

const car=document.getElementById("car");
const enemy=document.getElementById("enemy");
const coin=document.getElementById("coin");
const road=document.getElementById("road");

function move(dir) {
 if(!playing) return;
 x=Math.max(5,Math.min(235,x+dir*35));
 car.style.left=x+"px";
}

function startGame() {
 clearInterval(timer);
 x=120; y=-50; cy=-150; score=0;
 playing=true;
 car.style.left=x+"px";
 enemy.style.top=y+"px";
 coin.style.top=cy+"px";
 document.getElementById("score").innerText=0;
 document.getElementById("msg").innerText="Race started!";
 timer=setInterval(update,35);
}

function update() {
 y+=5;
 cy+=4;

 if(y>430) {
   y=-50;
   enemy.style.left=[20,120,220][Math.floor(Math.random()*3)]+"px";
 }

 if(cy>430) {
   cy=-100;
   coin.style.left=[30,120,210][Math.floor(Math.random()*3)]+"px";
 }

 enemy.style.top=y+"px";
 coin.style.top=cy+"px";

 let ex=parseInt(enemy.style.left);
 let cx=parseInt(coin.style.left);

 if(y>340 && y<405 && Math.abs(ex-x)<35) {
   playing=false;
   clearInterval(timer);
   document.getElementById("msg").innerText="Game Over! Score: "+score;
   return;
 }

 if(cy>340 && cy<405 && Math.abs(cx-x)<35) {
   score+=10;
   cy=-100;
   document.getElementById("score").innerText=score;
 }
}

document.addEventListener("keydown",e=>{
 if(e.key==="ArrowLeft") move(-1);
 if(e.key==="ArrowRight") move(1);
});
</script>
